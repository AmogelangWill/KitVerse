
import { useState, useEffect } from "react";
import { useAuth } from "./AuthContext";

function WishList() {

    const { user } = useAuth();

    const [product, setProducts] = useState(null);

        
    useEffect(() => {

        const FetchItems = async () => {
            console.log("NEW WISHLIST CODE IS RUNNING");
            if (!user) return;

            try {
                 console.log("User:", user);
                 console.log("Token exists:", !!user?.token);
                const response = await fetch( `https://kitverse-production.up.railway.app/api/Cart/wishlist/${user.userId}`,
                                {
                                    headers: {
                                        Authorization: `Bearer ${user.token}`
                                    }
                                }
                            );

                console.log("Wishlist status:", response.status);

                if (!response.ok) {
                    throw new Error(
                        `Wishlist request failed: ${response.status}`
                    );
                }

                const data = await response.json();

                console.log("Wishlist data:", data);

                setProducts(data);

            } catch (error) {

                console.error("Could not fetch wishlist:", error);

            }
        };

        FetchItems();

    }, [user?.userId]);




    const RemoveItems = async (TshirtId) => {

        if (!user) return;

        try {

            const Response = await fetch(
                `https://kitverse-production.up.railway.app/api/Cart/Wishlist/${user.userId}/${TshirtId}`,
                {
                    method: "DELETE"
                }
            );

            if (!Response.ok) {
                throw new Error(
                    `Failed to remove item: ${Response.status}`
                );
            }

            setProducts(previousItems => ({
                ...previousItems,

                itemLists: previousItems.itemLists.filter(
                    item => item.tShirt.tShirtId !== TshirtId
                )
            }));

        } catch (error) {

            console.error("Could not remove an item:", error);

        }
    };


    return (

        <div className="Wishlist_Container">

            {product?.itemLists?.map(item => (

                <div
                    key={item.tShirt.tShirtId}
                    className="P_Container"
                >

                    <p className="P_Description">
                        {item.tShirt?.team?.teamName}{" "}
                        {item.tShirt?.shirtSeason}
                    </p>

                    <p className="P_Price">
                        R {item.unitPurchaseAtTime?.toFixed(2)}
                    </p>

                    <img
                        className="PImages"
                        src={item.tShirt?.shirtLink}
                        alt="Product"
                    />

                    <button
                        className="RemoveItem"
                        onClick={() =>
                            RemoveItems(item.tShirt.tShirtId)
                        }
                    >
                        Remove
                    </button>

                </div>

            ))}

        </div>

    );
}

export default WishList;
